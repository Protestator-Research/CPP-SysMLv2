cmake_minimum_required(VERSION 3.15)

project(sysmlv2check CXX C)

# sysmlv2check: checker for SysML v2 texts (syntax, name resolution against the embedded standard library, structure,
# required elements). Options: BUILD_WITH_CHECK (root CMakeLists.txt); the tests need BUILD_TESTING.
#   sysmlv2check       library with the C++ class SysMLv2::Check::Checker and the C interface sysml_pruefe / sysml_freigeben
#   sysmlv2check-cli   the command line program `sysmlv2check` (stdin JSON -> stdout JSON)
#   sysmlv2check-test  the tests (gtest)
# For a WebAssembly build with Emscripten see cmake/Emscripten.cmake, cmake/emscripten.profile and the preset "emscripten-check" (check/cmake/EmscriptenPresets.json)
# (emsdk 5.0.2; see the README, section WebAssembly).

find_package(nlohmann_json REQUIRED)
find_package(Boost REQUIRED)

set(GLOBAL_HEADER
        include/sysmlv2/check/sysmlv2check_global.h)

set(API_HEADER
        include/sysmlv2/check/Checker.h
        include/sysmlv2/check/sysml_pruefe.h)

add_library(sysmlv2check
        ${GLOBAL_HEADER}
        ${API_HEADER}
        src/Checker.cpp
        src/CApi.cpp)

target_link_libraries(sysmlv2check
        Boost::headers
        nlohmann_json::nlohmann_json
        sysmlv2parser
        sysmlv2resources
        kerml
        sysmlv2interfaces)

target_include_directories(sysmlv2check PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:include>)
target_include_directories(sysmlv2check PRIVATE
        ../filehandling/include
        ../kerml/include
        ../sysmlinterfaces/include)

if(MSVC)
    target_compile_options(sysmlv2check PRIVATE /W4 /WX- /bigobj)
    target_link_directories(sysmlv2check PUBLIC ${CMAKE_BINARY_DIR}/out)
else()
    target_compile_options(sysmlv2check PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${SYSMLV2_WERROR}>:-Werror>)
endif()

# STRUKTUR diagnoses (LOGIK) come from KerML::Entities::validateRepresentation() (kerml/model/Representation.h, generated by
# kerml/tools/generate_abstract_syntax.py): violated multiplicities, missing mandatory references, null/duplicate entries of the
# elements of the user's text (derived properties are not checked). ON by default; OFF leaves the check out of the library.
option(SYSMLV2CHECK_VALIDATE_REPRESENTATION "sysmlv2check: report violations found by KerML::Entities::validateRepresentation (LOGIK/STRUKTUR)" ON)
if(SYSMLV2CHECK_VALIDATE_REPRESENTATION)
    target_compile_definitions(sysmlv2check PRIVATE SYSMLV2CHECK_STRUCTURE_CHECK)
endif()

if(NOT EMSCRIPTEN)
    # (with Emscripten the module sysmlv2check-wasm takes the output name; the command line program has no use there)
    add_executable(sysmlv2check-cli src/main.cpp)
    set_target_properties(sysmlv2check-cli PROPERTIES OUTPUT_NAME sysmlv2check)
    target_link_libraries(sysmlv2check-cli sysmlv2check)
endif()

include(GNUInstallDirs)
install(TARGETS sysmlv2check ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(TARGETS sysmlv2check RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(NOT EMSCRIPTEN)
    install(TARGETS sysmlv2check-cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(FILES ${GLOBAL_HEADER} ${API_HEADER} DESTINATION include/sysmlv2/check)

if(EMSCRIPTEN)
    include(cmake/Emscripten.cmake)
endif()

if(BUILD_TESTING)
    find_package(GTest REQUIRED)
    add_executable(sysmlv2check-test tests/TestCheck.cpp)
    target_link_libraries(sysmlv2check-test sysmlv2check gtest::gtest)
    target_include_directories(sysmlv2check-test PRIVATE
            ../filehandling/include
            ../kerml/include)
    add_dependencies(sysmlv2check-test sysmlv2check-cli)
    if(SYSMLV2CHECK_VALIDATE_REPRESENTATION)
        target_compile_definitions(sysmlv2check-test PRIVATE SYSMLV2CHECK_STRUCTURE_CHECK)
    endif()
    target_compile_definitions(sysmlv2check-test PRIVATE SYSMLV2CHECK_CLI="$<TARGET_FILE:sysmlv2check-cli>")
    add_test(NAME sysmlv2check COMMAND sysmlv2check-test)
endif()
